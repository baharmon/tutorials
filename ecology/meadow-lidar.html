Lidar

Dependencies

%conda install gdown

Setup

# Import modules
import sys
import subprocess
from pathlib import Path
from contextlib import suppress
import gdown

# Find GRASS Python packages
sys.path.append(
  subprocess.check_output(
    ["grass", "--config", "python_path"],
    text=True
    ).strip()
  )

# Import GRASS packages
import grass.script as gs
import grass.jupyter as gj
from grass.tools import Tools

# Create and initialize GRASS project
try:
    gj.init("hilltop")
except ValueError:
    with suppress(ValueError):
        gs.create_project("hilltop", epsg="2801")
        gj.init("hilltop")
tools = Tools(overwrite=True)

# Print projection
projection = tools.g_proj(format="shell", flags="p").keyval
print(projection.get("name"))
NAD83(HARN) / Louisiana South

Bin lidar

%%time

# Bin lidar
tools.r_in_pdal(
    input="meadow.laz",
    output="surface",
    method="max",
    resolution=0.3,
    flags="o"
    )

# Set color gradient
tools.r_colors(map="surface", color="viridis")

# Visualize
m = gj.Map(width=1600)
m.d_rast(map="surface")
m.d_legend(raster="surface", color="white", at=(50, 95, 2, 4))
m.show()

Import lidar

%%time

# Import lidar

    input="meadow.laz",
    output="lidar",
    flags="o"
    )

# Print point count

Decimate lidar
